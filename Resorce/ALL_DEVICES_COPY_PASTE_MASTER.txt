! --- IPSec VPN (HQ <-> BR01 Chattogram Demo Tunnel) ---

exit

 mode tunnel
exit
ip access-list extended ACL-IPSEC-VPN
 permit ip 10.10.0.0 0.0.255.255 10.1.1.0 0.0.0.255
exit
crypto map CMAP-GLOBALTECH 10 ipsec-isakmp
 set peer 172.16.1.2
 set transform-set TS-GLOBALTECH
 match address ACL-IPSEC-VPN
exit

! --- OSPF ---

! --- NAT/PAT ---

exit
